cmake_minimum_required(VERSION 3.20)
project(anime_lingo_enhance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(glslang CONFIG REQUIRED)
find_package(ncnn CONFIG REQUIRED)
find_package(Vulkan REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(shader_headers)
foreach(shader realesrgan_preproc realesrgan_postproc realesrgan_preproc_tta realesrgan_postproc_tta)
  foreach(mode plain fp16s int8s)
    set(defines)
    set(suffix "")
    if(NOT mode STREQUAL plain)
      list(APPEND defines -DNCNN_fp16_storage=1)
      set(suffix _${mode})
    endif()
    if(mode STREQUAL int8s)
      list(APPEND defines -DNCNN_int8_storage=1)
    endif()
    set(header ${CMAKE_CURRENT_BINARY_DIR}/${shader}${suffix}.spv.hex.h)
    add_custom_command(OUTPUT ${header}
      COMMAND ${GLSLANG_VALIDATOR} -V -s -x ${defines} -o ${header} ${CMAKE_CURRENT_SOURCE_DIR}/upstream/${shader}.comp
      DEPENDS upstream/${shader}.comp VERBATIM)
    list(APPEND shader_headers ${header})
  endforeach()
endforeach()
add_executable(anime-lingo-enhance enhance_worker.cpp upstream/realesrgan.cpp ${shader_headers})
target_include_directories(anime-lingo-enhance PRIVATE upstream ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(anime-lingo-enhance PRIVATE ncnn Vulkan::Vulkan)
target_compile_options(anime-lingo-enhance PRIVATE -O3)
